*-> proto(any) id="any" desc="generic node template: lang, type and name are required" act=[crt,exam] type=rule:
   -> lang: any requ
      type: any requ
      type==[dcm,file]:
      -> name: any requ

*-> proto(if) id="if" desc="if gate: single-instruction condition on singl, else object-object expression" act=[crt,exam]
   if(inst is singl): 
   -> stx="obj~obj"
   else
   -> stx="inst(obj~obj)"

*-> proto(goal) id="goal" desc="goal handler: prints context and success message" act=[wrt,exam,crt]
   show(cxt)
   show(msg(goal))

*-> proto(fail) id="fail" desc="fail handler: prints error, requests user approval with retry on wrt/crt" act=[wrt,exam,crt]
   show(err)
   if(act==[wrt,crt]){
      usr:apr:retry
   }   

*-> proto(varn) id="varn5" desc="variants gate: offers up to 5 options and requires user selection" type=rule scp=root
   show(varn:max(5)||req) ret=usr:sel:->

*-> proto(retry) id="r5" desc="retry loop: retries up to 5 times, asking the user when possible" type=rule scp=root
      for(var i=0;i<5;i=inc(1))<>:
      -> retry
      -> show(llm->user req?)<>:
         -> retry
         -> fail
